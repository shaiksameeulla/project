public void deleteNthNodeFromEnd(int n)
		    {
			 //Take two pointers – fast and slow. And initialize their values as head node
		        Node fast = head;
		        Node slow = head;

		        //Iterate fast pointer till the value of n.
		        for (int i = 0; i < n; i++) {
		            fast = fast.next;
		        }

		        if (fast == null) {
		              head = head.next;
		            return;
		        }
       // Now, start iteration of fast pointer till the None value of the linked list. Also, iterate slow pointer.
		        while (fast.next != null) {
		            fast = fast.next;
		            slow = slow.next;
		        }
		        /**
		         * Hence, once the fast pointer will reach to the end the slow pointer 
		         * will reach the node which you want to delete.
               Replace the next node of the slow pointer
                with the next to next node of the slow pointer
		         */
		        slow.next = slow.next.next;
		        return;
		    }
